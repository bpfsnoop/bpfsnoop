# Find a stable kernel function and verify structured function metadata.
name: mcp::find
tag: mcp,find
tool: find
arguments: {"pattern":"vfs_read","kind":"function","limit":1}

# Verify that an invalid kernel address is rejected through MCP.

match: failed to read kernel data

timeout: 15s

# Repeated reads ensure the trigger remains active until attachment completes.
name: mcp::trace
tag: mcp,trace
tool: trace
arguments: {"targets":[{"kind":"function","name":"vfs_read"}],"filter":{"comm":"dd"},"capture":{"arguments":true,"retval":true},"limits":{"duration_ms":10000,"max_events":1}}
match: "comm":"dd"
timeout: 30s
trigger: while true; do dd if=/proc/version of=/dev/null bs=1 count=1 >/dev/null 2>&1; done

---

# Abort a trace whose filter deliberately cannot match an event.
name: mcp::abort
tag: mcp,abort,trace
tool: trace
arguments: {"targets":[{"kind":"function","name":"vfs_read"}],"filter":{"comm":"never-match"},"limits":{"duration_ms":30000,"max_events":1}}
match: "stopped_by":"abort"
abort-after: 2s
timeout: 20s
