# Filter a tracepoint with a C argument expression.
name: mcp::trace:filter-arg
tag: mcp,trace,filter-arg,tracepoint
tool: trace
arguments: {"targets":[{"kind":"tracepoint","name":"netif_receive_skb"}],"filter":{"expr":"1 == skb->dev->ifindex"},"capture":{"arguments":true},"limits":{"duration_ms":10000,"max_events":1}}
match: "kind":"tracepoint"
timeout: 30s
trigger: while true; do ping -c 1 -s 64 127.0.0.1 >/dev/null 2>&1; done
