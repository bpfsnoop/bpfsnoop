# Rebuild xdpcrc before running: ./localtest --test-file t/cc/bpf_map.txt
# xdpcrc creates cc_lookup with BTF and key lo.ifindex, holding
# {count: 42, nest: {count: 84}, tail: 7}. The prerequisite owns its lifetime.
# Run these tests serially without another xdpcrc process.

# Lookup by name with the key size inferred from the map key BTF.
name: cc::map-lookup:name
tag: cc,builtin,map,pkt,tp
prerequisite: ./xdpcrc -d lo
prerequisite-ready: Attached xdp to lo
test: -t netif_receive_skb --filter-pkt 'host 127.0.0.1 and icmp' --filter-arg 'map_lookup("cc_lookup", &skb->dev->ifindex)->count == 42' --output-arg 'map_lookup("cc_lookup", &skb->dev->ifindex)->count' --limit-events 1
match: 'map_lookup("cc_lookup", &skb->dev->ifindex)->count'=0x2a/42
trigger: ping -c 1 -s 64 127.0.0.1
timeout: 10s

# Wait for the prerequisite map before resolving its ID; requires bpftool.
name: cc::map-lookup:id
tag: cc,builtin,map,pkt,tp
prerequisite: ./xdpcrc -d lo
prerequisite-ready: Attached xdp to lo
test: -t netif_receive_skb --filter-pkt 'host 127.0.0.1 and icmp' --filter-arg "map_lookup($(for attempt in {1..50}; do map_id=$(bpftool map show name cc_lookup | head -n 1 | cut -d: -f1); if [ -n "$map_id" ]; then printf '%s' "$map_id"; break; fi; sleep 0.1; done), &skb->dev->ifindex)->count == 42" --output-arg 'skb->dev->ifindex' --limit-events 1
match: (int)'skb->dev->ifindex'=1
trigger: ping -c 1 -s 64 127.0.0.1
timeout: 10s
