# These tests are for tracing XDP programs in the kernel.

# install XDP program
name: xdp::xdp_install
tag: xdp,fnarg
test: -k generic_xdp_install --output-arg 'xdp->prog->aux->id' --output-arg 'xdp->prog->aux->name' --output-arg 'xdp->prog->aux->used_maps[0]->map_type' --output-arg 'xdp->prog->aux->used_maps[0]->name' --output-arg 'dev->ifindex' --output-arg 'dev->name'
match: (array(char[16]))'xdp->prog->aux->name'="crc"
trigger: ./xdpcrc -d lo

---

# capture packets
name: xdp::capture_packets
tag: xdp,pkt,fnarg
test: -p 'n:crc' --filter-pkt 'host 127.0.0.1 and icmp' --output-arg '*ctx' --output-arg 'buf(ctx->data, 34)'
match: (struct xdp_buff)'*ctx'={"data":
prerequisite: ./xdpcrc -d lo
trigger: ping -c 1 -s 64 127.0.0.1

---

# both entry and exit mode
name: xdp::entry_exit
tag: xdp,pkt,fnarg
test: -p 'n:crc' --filter-pkt 'host 127.0.0.1 and icmp' --output-arg '*ctx' --output-arg 'buf(ctx->data, 34)' -m 'entry,exit' -v
match: → crc[bpf]
prerequisite: ./xdpcrc -d lo
trigger: ping -c 1 -s 64 127.0.0.1

---

# retval of a bpf subprog whose BTF the verifier marks unreliable
name: xdp::subprog_retval
tag: xdp,retval
test: -p 'n:crc:retval_probe' -m exit --filter-arg '(int)$retval == 42' --limit-events 1
match: ← retval_probe[bpf]
prerequisite: ./xdpcrc -d lo
trigger: ping -c 1 -s 64 127.0.0.1

---

# the same subprog seen through the function graph of its prog
name: xdp::subprog_retval:fgraph
tag: xdp,retval,fngraph
test: -p 'n:crc' -m exit --output-fgraph --fgraph-max-depth 1 --limit-events 1
match: (41)) retval=42
prerequisite: ./xdpcrc -d lo
trigger: ping -c 1 -s 64 127.0.0.1

---

# indexing an array member loads the element, not its address
name: xdp::index:array
tag: xdp,fnarg,index
test: -k generic_xdp_install --output-arg 'dev->name[0]' --output-arg 'dev->name[1]'
match: (char)'dev->name[0]'=0x6c/108, (char)'dev->name[1]'=0x6f/111
trigger: ./xdpcrc -d lo

---

# indexing a pointer to struct, then a member of the element
name: xdp::index:ptr_struct
tag: xdp,fnarg,index
test: -k generic_xdp_install --output-arg 'str(dev->_tx[0].dev->name)'
match: 'str(dev->_tx[0].dev->name)'="lo"
trigger: ./xdpcrc -d lo

---

# indexing a pointer arg of a bpf subprog
name: xdp::index:subprog_ptr
tag: xdp,fnarg,index
test: -p 'n:crc:retval_probe' -m entry --output-arg 'v[0]' --limit-events 1
match: (int)'v[0]'=41
prerequisite: ./xdpcrc -d lo
trigger: ping -c 1 -s 64 127.0.0.1
