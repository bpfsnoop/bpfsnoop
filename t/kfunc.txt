# These tests are for tracing kfuncs in the kernel.

# icmp_rcv
name: kfuncs::icmp_rcv
tag: kfuncs,pkt,fnarg
test: -k 'icmp_rcv' --filter-pkt 'host 127.0.0.1' --output-pkt --output-arg 'skb->dev->name' -m 'entry,exit' -v
match: → icmp_rcv
trigger: ping -c 1 -s 64 127.0.0.1

---

# Capture a kernel stack after selected arguments and packet data.
name: kfuncs::icmp_rcv:output-stack
tag: kfuncs,pkt,fnarg,output-stack
test: -k 'icmp_rcv' --filter-pkt 'icmp and host 127.0.0.1' --output-arg 'skb->cb[0]' --output-pkt --output-stack
match: ip_local_deliver+0x
trigger: ping -c 1 -s 64 127.0.0.1

---

# Output the LBR stack, which requires LBR hardware.
name: kfuncs::icmp_rcv:output-lbr
tag: kfuncs,lbr
feature: lbr
test: -k 'icmp_rcv' --output-lbr
match: LBR stack:
trigger: ping -c 1 -s 64 127.0.0.1

---
